### 2. schema.prisma — Better Auth tables (required by Better Auth Prisma adapter)
```prisma
model User {
  id            String   @id @default(cuid())
  name          String
  email         String   @unique
  emailVerified Boolean  @default(false)
  image         String?
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt
  sessions      Session[]
  accounts      Account[]
  // Business relation
  agentInstances AgentInstance[]
  auditLogs     AuditLog[]
}

model Session {
  id        String   @id @default(cuid())
  expiresAt DateTime
  token     String   @unique
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  ipAddress String?
  userAgent String?
  userId    String
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
}

model Account {
  id                   String    @id @default(cuid())
  accountId            String
  providerId           String
  userId               String
  user                 User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  accessToken          String?
  refreshToken         String?
  idToken              String?
  accessTokenExpiresAt DateTime?
  refreshTokenExpiresAt DateTime?
  scope                String?
  password             String?
  createdAt            DateTime  @default(now())
  updatedAt            DateTime  @updatedAt
}

model Verification {
  id         String   @id @default(cuid())
  identifier String
  value      String
  expiresAt  DateTime
  createdAt  DateTime @default(now())
  updatedAt  DateTime @updatedAt
}
```

### 3. schema.prisma — Business tables
```prisma
model Workspace {
  id             String          @id @default(cuid())
  name           String
  createdAt      DateTime        @default(now())
  updatedAt      DateTime        @updatedAt
  agentInstances AgentInstance[]
  integrations   Integration[]
  executions     Execution[]
  notifications  Notification[]
  auditLogs      AuditLog[]
}

model AgentTemplate {
  id                       String          @id
  name                     String
  category                 String
  description              String?
  taskScope                String?
  limitationsJson          Json            @default("[]")
  requiredIntegrationsJson Json            @default("[]")
  optionalIntegrationsJson Json            @default("[]")
  allowedTriggerModesJson  Json            @default("[]")
  configSchemaJson         Json            @default("{}")
  outputContractJson       Json            @default("{}")
  n8nWebhookPath           String
  approvalRequired         Boolean         @default(false)
  costLevel                String          @default("low")
  status                   String          @default("active")
  createdAt                DateTime        @default(now())
  updatedAt                DateTime        @updatedAt
  agentInstances           AgentInstance[]
  executions               Execution[]
}

model AgentInstance {
  id                    String                    @id @default(cuid())
  workspaceId           String
  workspace             Workspace                 @relation(fields: [workspaceId], references: [id])
  templateId            String
  template              AgentTemplate             @relation(fields: [templateId], references: [id])
  name                  String
  status                String                    @default("draft")
  triggerMode           String                    @default("manual")
  configJson            Json                      @default("{}")
  createdById           String
  createdBy             User                      @relation(fields: [createdById], references: [id])
  createdAt             DateTime                  @default(now())
  updatedAt             DateTime                  @updatedAt
  integrations          AgentInstanceIntegration[]
  executions            Execution[]
  notifications         Notification[]
}

model Integration {
  id              String                    @id @default(cuid())
  workspaceId     String
  workspace       Workspace                 @relation(fields: [workspaceId], references: [id])
  integrationKey  String
  name            String
  category        String?
  status          String                    @default("mock_connected")
  mockEnabled     Boolean                   @default(true)
  lastTestedAt    DateTime?
  metadataJson    Json                      @default("{}")
  createdAt       DateTime                  @default(now())
  updatedAt       DateTime                  @updatedAt
  agentInstances  AgentInstanceIntegration[]
}

model AgentInstanceIntegration {
  id              String        @id @default(cuid())
  agentInstanceId String
  agentInstance   AgentInstance @relation(fields: [agentInstanceId], references: [id], onDelete: Cascade)
  integrationId   String
  integration     Integration   @relation(fields: [integrationId], references: [id])
  createdAt       DateTime      @default(now())
}

model Execution {
  id              String           @id @default(cuid())
  workspaceId     String
  workspace       Workspace        @relation(fields: [workspaceId], references: [id])
  agentInstanceId String
  agentInstance   AgentInstance    @relation(fields: [agentInstanceId], references: [id])
  templateId      String
  template        AgentTemplate    @relation(fields: [templateId], references: [id])
  status          String           @default("queued")
  triggerType     String           @default("manual")
  startedAt       DateTime?
  finishedAt      DateTime?
  durationMs      Int?
  inputCount      Int              @default(0)
  outputCount     Int              @default(0)
  errorCount      Int              @default(0)
  tokensInput     Int              @default(0)
  tokensOutput    Int              @default(0)
  estimatedCostUsd Decimal         @default(0)
  n8nExecutionId  String?
  resultSummary   String?
  errorMessage    String?
  createdAt       DateTime         @default(now())
  updatedAt       DateTime         @updatedAt
  output          ExecutionOutput?
  notifications   Notification[]
}

model ExecutionOutput {
  id          String    @id @default(cuid())
  executionId String    @unique
  execution   Execution @relation(fields: [executionId], references: [id], onDelete: Cascade)
  outputJson  Json      @default("{}")
  createdAt   DateTime  @default(now())
}

model Notification {
  id              String        @id @default(cuid())
  workspaceId     String
  workspace       Workspace     @relation(fields: [workspaceId], references: [id])
  agentInstanceId String
  agentInstance   AgentInstance @relation(fields: [agentInstanceId], references: [id])
  executionId     String
  execution       Execution     @relation(fields: [executionId], references: [id])
  severity        String
  type            String
  title           String
  message         String?
  status          String        @default("unread")
  createdAt       DateTime      @default(now())
  updatedAt       DateTime      @updatedAt
}

model AuditLog {
  id          String    @id @default(cuid())
  workspaceId String
  workspace   Workspace @relation(fields: [workspaceId], references: [id])
  actorUserId String
  actor       User      @relation(fields: [actorUserId], references: [id])
  entityType  String
  entityId    String
  action      String
  beforeJson  Json?
  afterJson   Json?
  createdAt   DateTime  @default(now())
}
```
