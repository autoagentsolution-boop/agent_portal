### 2. use-notifications.ts hook
```typescript
function useNotifications(filters?: { status?: string; severity?: string }) {
  return useQuery({
    queryKey: ['notifications', filters],
    queryFn: () => apiClient.get('/notifications', { params: filters }),
  });
}

function useNotificationMutations() {
  const queryClient = useQueryClient();
  const invalidate = () => queryClient.invalidateQueries({ queryKey: ['notifications'] });
  
  const acknowledge = useMutation({ mutationFn: (id) => apiClient.patch(`/notifications/${id}/acknowledge`), onSuccess: invalidate });
  const resolve = useMutation({ mutationFn: (id) => apiClient.patch(`/notifications/${id}/resolve`), onSuccess: invalidate });
  return { acknowledge, resolve };
}
```
